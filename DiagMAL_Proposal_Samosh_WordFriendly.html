1.4 Machine Learning Implementation

Outbreak Prediction Model

import tensorflow as tf import numpy as np from sklearn.preprocessing import StandardScaler from tensorflow.keras import layers, models class MalariaOutbreakPredictor: def __init__(self): self.model = self.build_st_gnn() self.scaler = StandardScaler() def build_st_gnn(self): """Build Spatio-Temporal Graph Neural Network""" # Input layers spatial_input = layers.Input(shape=(None, 128)) # Spatial features temporal_input = layers.Input(shape=(30, 64)) # 30-day temporal window # Spatial processing with Graph Attention Networks gat_layer = layers.MultiHeadAttention( num_heads=8, key_dim=64 )(spatial_input, spatial_input) # Temporal processing with LSTM lstm_layer = layers.LSTM(128, return_sequences=True)(temporal_input) temporal_features = layers.GlobalMaxPooling1D()(lstm_layer) # Combine spatial and temporal features combined = layers.Concatenate()([ layers.GlobalMaxPooling1D()(gat_layer), temporal_features ]) # Dense layers for prediction dense1 = layers.Dense(256, activation='relu')(combined) dropout1 = layers.Dropout(0.3)(dense1) dense2 = layers.Dense(128, activation='relu')(dropout1) dropout2 = layers.Dropout(0.3)(dense2) # Output layer (outbreak probability) output = layers.Dense(1, activation='sigmoid')(dropout2) model = models.Model( inputs=[spatial_input, temporal_input], outputs=output ) model.compile( optimizer='adam', loss='binary_crossentropy', metrics=['accuracy', 'AUC'] ) return model def predict_outbreak(self, spatial_data, temporal_data, threshold=0.7): """Predict malaria outbreak probability""" # Preprocess data spatial_normalized = self.scaler.fit_transform(spatial_data) # Make prediction probability = self.model.predict([spatial_normalized, temporal_data]) # Generate alerts if probability exceeds threshold if probability > threshold: self.trigger_alert(probability, spatial_data) return { 'outbreak_probability': float(probability), 'risk_level': self.categorize_risk(probability), 'recommended_actions': self.get_recommendations(probability) }

Clustering for Hotspot Detection

from sklearn.cluster import DBSCAN import geopandas as gpd from shapely.geometry import Point class MalariaHotspotDetector: def __init__(self, eps_km=5, min_samples=10): """ Initialize DBSCAN clustering for hotspot detection eps_km: Maximum distance between cases (in kilometers) min_samples: Minimum cases to form a cluster """ self.eps = eps_km / 111 # Convert km to degrees (approximate) self.min_samples = min_samples def detect_hotspots(self, case_data): """Detect malaria hotspots using spatial clustering""" # Extract coordinates coordinates = case_data[['latitude', 'longitude']].values # Apply DBSCAN clustering clustering = DBSCAN( eps=self.eps, min_samples=self.min_samples, metric='haversine' ).fit(np.radians(coordinates)) # Add cluster labels to data case_data['cluster'] = clustering.labels_ # Identify hotspots (non-noise clusters) hotspots = case_data[case_data['cluster'] != -1] # Calculate hotspot statistics hotspot_stats = hotspots.groupby('cluster').agg({ 'case_id': 'count', 'latitude': 'mean', 'longitude': 'mean', 'severity': 'mean' }).rename(columns={'case_id': 'case_count'}) return self.generate_hotspot_report(hotspot_stats) def generate_hotspot_report(self, hotspot_stats): """Generate actionable hotspot report""" report = { 'total_hotspots': len(hotspot_stats), 'high_risk_areas': [], 'intervention_recommendations': [] } for idx, row in hotspot_stats.iterrows(): if row['case_count'] > 20: # High-risk threshold report['high_risk_areas'].append({ 'location': (row['latitude'], row['longitude']), 'cases': row['case_count'], 'average_severity': row['severity'], 'recommended_action': 'IMMEDIATE_INTERVENTION' }) # Generate specific recommendations report['intervention_recommendations'].append( self.get_intervention_plan(row) ) return report

# Automated Model Monitoring and Retraining Pipeline import mlflow import pandas as pd from sklearn.metrics import roc_auc_score, precision_recall_curve from datetime import datetime, timedelta class ModelMonitoringPipeline: def __init__(self): self.mlflow_tracking_uri = "http://mlflow.diagmal.ao" self.performance_threshold = 0.75 self.drift_threshold = 0.15 def monitor_model_performance(self, model_name, production_data): """Continuous monitoring of model performance in production""" # Load current production model client = mlflow.tracking.MlflowClient() model_version = client.get_latest_versions( model_name, stages=["Production"] )[0] model = mlflow.pyfunc.load_model( f"models:/{model_name}/{model_version.version}" ) # Generate predictions predictions = model.predict(production_data.features) # Calculate performance metrics auc_score = roc_auc_score( production_data.labels, predictions ) # Check for model drift drift_score = self.calculate_drift( production_data, model_version.run_id ) # Log metrics with mlflow.start_run(): mlflow.log_metric("production_auc", auc_score) mlflow.log_metric("data_drift", drift_score) mlflow.log_metric("prediction_latency", self.measure_latency(model, production_data)) # Trigger retraining if needed if auc_score < self.performance_threshold or drift_score > self.drift_threshold: self.trigger_retraining(model_name, production_data) return { 'model_version': model_version.version, 'auc_score': auc_score, 'drift_score': drift_score, 'requires_retraining': auc_score < self.performance_threshold, 'timestamp': datetime.now().isoformat() }

6.2 Data Quality Assurance

# Automated Data Quality Checks class DataQualityMonitor: def __init__(self): self.quality_rules = { 'completeness': 0.95, # 95% fields must be complete 'accuracy': 0.98, # 98% accuracy threshold 'timeliness': 24, # Data must be <24 hours old 'consistency': 0.99 # 99% consistency across sources } def validate_incoming_data(self, data_batch): """Real-time data quality validation""" quality_report = { 'batch_id': data_batch.id, 'timestamp': datetime.now(), 'checks': {} } # Completeness check completeness_score = ( data_batch.count() - data_batch.isna().sum() ) / data_batch.count() quality_report['checks']['completeness'] = { 'score': completeness_score, 'passed': completeness_score >= self.quality_rules['completeness'], 'missing_fields': self.identify_missing_fields(data_batch) } # Accuracy check - validate against known patterns accuracy_score = self.validate_accuracy(data_batch) quality_report['checks']['accuracy'] = { 'score': accuracy_score, 'passed': accuracy_score >= self.quality_rules['accuracy'], 'anomalies': self.detect_anomalies(data_batch) } # Timeliness check data_age = (datetime.now() - data_batch['timestamp'].max()).hours quality_report['checks']['timeliness'] = { 'age_hours': data_age, 'passed': data_age <= self.quality_rules['timeliness'] } # Consistency check across multiple sources consistency_score = self.check_cross_source_consistency(data_batch) quality_report['checks']['consistency'] = { 'score': consistency_score, 'passed': consistency_score >= self.quality_rules['consistency'], 'conflicts': self.identify_conflicts(data_batch) } # Take action based on quality if not all(check['passed'] for check in quality_report['checks'].values()): self.handle_quality_issues(quality_report) return quality_report
